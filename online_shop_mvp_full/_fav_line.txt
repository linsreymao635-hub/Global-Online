class _Fav extends State<FavoritePage>{@override Widget build(BuildContext c)=>Scaffold(appBar:AppBar(title:const Text('Favorites')),body:FutureBuilder<List<Product>>(future:widget.source,builder:(_,s){if(!s.hasData)return const Center(child:CircularProgressIndicator());final list=widget.fav.ids.isEmpty?<Product>[]:s.data!.where((p)=>widget.fav.has(p)).toList();return list.isEmpty?const Center(child:Text('No favorites')):ListView(children:list.map((p)=>ListTile(leading:Image.network(p.thumbnail,width:60),title:Text(p.title),trailing:IconButton(icon:const Icon(Icons.favorite),onPressed:()=>setState(()=>widget.fav.toggle(p)))).toList());}));}}
